"""Local-only GUI server. No deployment, accounts, remote access or external telemetry."""

from __future__ import annotations

import argparse
import json
from http.server import BaseHTTPRequestHandler, HTTPServer
from importlib.resources import files

from physical_fpv.fpv.simulator import Simulator


def make_server(port=8765, *, frame_density_kg_m3=None):
    simulator = Simulator(frame_density_kg_m3=frame_density_kg_m3)

    class Handler(BaseHTTPRequestHandler):
        def respond(self, status, payload, content_type="application/json"):
            data = (
                payload.encode()
                if isinstance(payload, str)
                else json.dumps(payload, allow_nan=False).encode()
            )
            self.send_response(status)
            self.send_header("Content-Type", content_type)
            self.send_header("Content-Length", str(len(data)))
            self.send_header("Cache-Control", "no-store")
            self.send_header("X-Content-Type-Options", "nosniff")
            self.end_headers()
            self.wfile.write(data)

        def do_GET(self):
            if self.path == "/":
                self.respond(
                    200,
                    files("physical_fpv.fpv")
                    .joinpath("web/index.html")
                    .read_text(encoding="utf-8"),
                    "text/html; charset=utf-8",
                )
            elif self.path == "/state":
                self.respond(200, simulator.telemetry())
            elif self.path == "/replay":
                self.respond(200, simulator.replay_bundle())
            else:
                self.respond(404, {"error": "Not found"})

        def do_POST(self):
            # Reject cross-origin writes; only local browser UI submits JSON.
            origin = self.headers.get("Origin")
            if origin and origin not in {
                f"http://127.0.0.1:{self.server.server_port}",
                f"http://localhost:{self.server.server_port}",
            }:
                self.respond(403, {"error": "Cross-origin requests are disabled"})
                return
            try:
                length = int(self.headers.get("Content-Length", "0"))
                if not 0 < length <= 4096:
                    raise ValueError("Invalid request size")
                if self.headers.get("Content-Type", "").split(";")[0] != "application/json":
                    raise ValueError("application/json required")
                body = json.loads(self.rfile.read(length))
                if not isinstance(body, dict):
                    raise ValueError("JSON object required")
                if self.path == "/step":
                    result = simulator.advance(
                        body.get("controls", {}), body.get("duration_s", 0.05)
                    )
                elif self.path == "/reset":
                    simulator.__init__(frame_density_kg_m3=frame_density_kg_m3)
                    result = simulator.telemetry()
                else:
                    self.respond(404, {"error": "Not found"})
                    return
                self.respond(200, result)
            except (ValueError, TypeError, KeyError) as exc:
                self.respond(400, {"error": str(exc)})

        def log_message(self, *args):
            pass

    return HTTPServer(("127.0.0.1", port), Handler)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--port", type=int, default=8765)
    parser.add_argument(
        "--emt-frame", action="store_true", help="Use bounded ASE Al density for frame mass"
    )
    args = parser.parse_args()
    density = None
    if args.emt_frame:
        from physical_fpv.fpv.atomistic import aluminum_eos

        density = aluminum_eos()["density_kg_m3"]
    server = make_server(args.port, frame_density_kg_m3=density)
    print(f"Local synthetic FPV GUI: http://127.0.0.1:{server.server_port}", flush=True)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()


if __name__ == "__main__":
    main()
