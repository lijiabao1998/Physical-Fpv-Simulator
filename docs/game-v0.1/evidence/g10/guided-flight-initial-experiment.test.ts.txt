import {it,expect} from 'vitest';import {writeFileSync} from 'node:fs';import {initPhysics} from '../src/physics/world';import {GameSession} from '../src/game/session';import {getCourse} from '../src/game/courses';import {rotate,clamp,euler} from '../src/physics/math';
it('exploratory input-only British route controller, not player autopilot',async()=>{await initPhysics();const s=new GameSession(undefined,true,undefined,'british-slalom');s.start();s.arm(0);let phase=0,gate=0,hover=s.power.hover();const trace=[];
for(let tick=0;tick<240*110&&s.state==='flying';tick++){const g=getCourse(s.courseId).gates[gate],n=rotate(g.rotation,[1,0,0]),p=s.world.position,v=s.world.velocity;const target=g.center.map((x,k)=>x+(phase===0?-3:3)*n[k]);if(phase===0&&Math.hypot(...target.map((x,k)=>x-p[k]))<.6)phase=1;if(s.course.next>gate){gate=s.course.next;phase=0;if(gate>=7)break;}const [roll,pitch,yaw]=euler(s.world.quaternion),a=target.map((x,k)=>clamp(1.3*(x-p[k])-2*v[k],-3,3));const ax=Math.cos(yaw)*a[0]+Math.sin(yaw)*a[1],ay=-Math.sin(yaw)*a[0]+Math.cos(yaw)*a[1];const desiredYaw=Math.atan2(n[1],n[0]),dy=Math.atan2(Math.sin(desiredYaw-yaw),Math.cos(desiredYaw-yaw));if(tick%120===0)hover=s.power.hover();const c={throttle:clamp(hover*Math.sqrt((9.80665+a[2])/(9.80665*Math.max(.4,Math.cos(roll)*Math.cos(pitch)))),.05,.8),pitch:clamp(Math.atan2(ax,9.80665+a[2])/.6,-1,1),roll:clamp(-Math.atan2(ay,9.80665+a[2])/.6,-1,1),yaw:clamp(dy*.8,-.4,.4)};s.step(c);if(tick%240===0)trace.push({tick,p,gate,phase,c});}
const end=s.snapshot(),bundle=s.exportReplay('local-exploration');expect(end.gate).toBe(7);expect(end.state).toBe('finished');s.replay(bundle);for(let i=0;i<bundle.events.length;i++)s.step({throttle:0,roll:0,pitch:0,yaw:0});expect(s.world.position).toEqual(end.position);expect(s.course.splits).toEqual(end.splits);
writeFileSync('../docs/game-v0.1/evidence/g10/guided-flight-exploration.json',JSON.stringify({scope:'Test-only controls through original motor mixer; no transforms/forces/physics modifications. Not human/controller validation.',end,trace,replay:bundle},null,2));s.world.dispose();});
