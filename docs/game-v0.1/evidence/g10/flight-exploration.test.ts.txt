import {writeFileSync} from 'node:fs';import {beforeAll,it,expect} from 'vitest';import {initPhysics} from '../src/physics/world';import {GameSession} from '../src/game/session';import {ZERO} from '../src/physics/controller';beforeAll(initPhysics);
it('G10 preserved fixed-input flight exploration without body-state writes',()=>{const results=[];for(const throttle of [.38,.42,.46,.50])for(const pitch of [.25,.4,.6,.8]){const s=new GameSession(undefined,true,undefined,'british-slalom');s.start();s.arm(0);for(let i=0;i<1440&&s.state==='flying';i++)s.step({...ZERO,throttle:i<180?.56:throttle,pitch:i<180?0:pitch});results.push({throttle,pitch,gates:s.course.next,position:s.world.position,state:s.state});s.world.dispose();}writeFileSync('/tmp/g10-flight-exploration.json',JSON.stringify(results,null,2));expect(results).toHaveLength(16);});
